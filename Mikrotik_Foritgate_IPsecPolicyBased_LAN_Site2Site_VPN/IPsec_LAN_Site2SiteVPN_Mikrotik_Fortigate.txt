+ Scenario Ex:
 - Fortigate
	+ WAN IP : 10.10.8.1/30 - Interface = port8-WAN
        + LAN IP : 172.16.7.1/24 - Interface = port7-LAN
 - Mikrotik 
	+ WAN IP : 10.10.8.2/30 - Interface = ether8-WAN
	+ LAN IP : 192.168.7.1/24 - Interface = ether7-LAN

Note : Before Config it Make Sure both Intermedia Can reach each other Public IP

+ Foritgate CLI Configure :

conf vpn ipsec phase1-interface
edit "Mikrotik-Tunnel-P1"
set int port8
set ike-version 2
set local-gw 10.10.8.1
set remote-gw 10.10.8.2
set net-device enable
set keylife 28800
set peertype any
set proposal des-sha256
set dpd on-idle
set dhgrp 14
set dpd-retryinterval 10
set psksecret MySecurePassword123
show ------- (Comfirm it output like you assign)
next
end


conf vpn ipsec phase2-interface
edit "Mikrotik-Tunnel-P2"
set phase1name "Mikrotik-Tunnel-P1"
set proposal des-sha256
set dhgrp 14
set keylifeseconds 3600
set src-subnet 172.16.7.0 255.255.255.0
set dst-subnet 192.168.7.0 255.255.255.0
show ------- (Confirm it output like you assign)
next
end


conf firewall address
edit "Fortigate-LAN"
set subnet 172.16.7.0 255.255.255.0
next
edit "Mikrotik-LAN"
set subnet 192.168.7.0 255.255.255.0
next
end

conf firewall policy
edit 0
set name "Out-To-Mikrotik"
set srcintf port7
set dstintf Mikrotik-Tunnel-P1
set srcaddr Fortigate-LAN
set dstaddr Mikrotik-LAN
set action accept
set schedule always
set service ALL
next
edit 0
set name "In-To-Mikrotik"
set srcintf Mikrotik-Tunnel-P1
set dstintf port7
set srcaddr Mikrotik-LAN
set dstaddr Fortigate-LAN
set action accept
set schedule always
set service ALL
next
show ------- (Confirm your rule that had assign)


conf router static
edit 1
set dst 192.168.7.0 255.255.255.0
set device Mikrotik-Tunnel-P1

+ Mikrotik CLI Configure :

ip/ipsec/profile

add name="Fortigate-Profile" enc-algorithm=aes-128,des hash-algorithm=sha256
dh-group=modp2048 nat-traversal=yes dpd-interval=10s dpd-maximum-failures=5
lifetime=8h

ip/ipsec/proposal

add name="Fortigate-Proposal" auth-algorithms=sha256 enc-algorithms=des
lifetime=1h pfs-group=modp2048

ip/ipsec/peer

add name="Fortigate-Peer" profile=Fortigate-Profile address=10.10.8.1 
exchange-mode=ike2

ip/ipsec/identity

add peer=Fortigate-Peer auth-method=pre-shared-key secret=MySecurePassword123
generate-policy=no

ip/ipsec/policy

add peer=Fortigate-Peer proposal=Fortigate-Proposal src-address=192.168.7.0/24
dst-address=172.16.7.0/24 tunnel=yes level=require action=encrypt

ip/firewall/filter

add chain=input dst-port=500,4500 protocol=udp action=accept comment="Allowing IKE/NAT-T"

add chain=input protocol=ipsec-esp action=accept comment="Allowing ESP"

ip/route

add dst-address=172.16.7.0/24 gateway=10.10.8.1 comment="To-Fortigate-LAN"
